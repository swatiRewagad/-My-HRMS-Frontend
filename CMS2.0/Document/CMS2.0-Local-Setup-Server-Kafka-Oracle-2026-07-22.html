3. Setup Keycloak (Local)

Keycloak runs locally since it needs to be accessible from your browser for OIDC redirects.

1 Start Keycloak

# Navigate to Keycloak installation
cd c:/tools/keycloak-26.0.0

# Start in dev mode on port 9090 (avoids conflict with API Gateway on 8080)
bin/kc.bat start-dev --http-port=9090

# Admin console: http://localhost:9090
# Default admin: admin / admin

2 Create Realm & Client

# Set kcadm credentials
bin/kcadm.bat config credentials --server http://localhost:9090 --realm master --user admin --password admin

# Create 'cms' realm
bin/kcadm.bat create realms -s realm=cms -s enabled=true

# Create 'cms-frontend' client (public OIDC client)
bin/kcadm.bat create clients -r cms -s clientId=cms-frontend -s enabled=true \
  -s publicClient=true -s directAccessGrantsEnabled=true \
  -s "redirectUris=[\"http://localhost:4200/*\",\"http://localhost:4201/*\"]" \
  -s "webOrigins=[\"http://localhost:4200\",\"http://localhost:4201\"]"

# Create 'admin-cli' client for backend (confidential)
bin/kcadm.bat create clients -r cms -s clientId=admin-cli -s enabled=true

4. Configure Backend for Server Kafka & Oracle

1 Create/Edit Application Profile

Create a new profile application-dev-server.yml that connects to server infrastructure while running locally.

File: cms-backend/src/main/resources/application-dev-server.yml

# ============================================================
# CMS Backend — Dev Server Profile
# Connects to server Kafka & Oracle while running locally
# Activate: mvn spring-boot:run -Dspring-boot.run.profiles=dev-server
# ============================================================

server:
  port: 8082

spring:
  datasource:
    # Oracle connection to server
    url: jdbc:oracle:thin:@<ORACLE_SERVER_IP>:1521/<SERVICE_NAME>
    username: <DB_USERNAME>
    password: <DB_PASSWORD>
    driver-class-name: oracle.jdbc.OracleDriver
    hikari:
      maximum-pool-size: 5
      minimum-idle: 2
      connection-timeout: 10000
      pool-name: CMS-Backend-DevServer-Pool
  jpa:
    hibernate:
      ddl-auto: validate  # NEVER use 'update' against shared server DB
    show-sql: true
    properties:
      hibernate:
        dialect: org.hibernate.dialect.OracleDialect
        format_sql: true
        jdbc:
          batch_size: 10
    open-in-view: false

  kafka:
    bootstrap-servers: <KAFKA_SERVER_IP>:9092
    producer:
      key-serializer: org.apache.kafka.common.serialization.StringSerializer
      value-serializer: org.apache.kafka.common.serialization.StringSerializer
      acks: all
      retries: 3
      properties:
        max.block.ms: 5000
        delivery.timeout.ms: 10000
        request.timeout.ms: 5000
    consumer:
      group-id: cms-backend-dev
      auto-offset-reset: latest
    listener:
      missing-topics-fatal: false

  cache:
    type: simple

keycloak:
  admin:
    server-url: http://localhost:9090
    realm: cms
    client-id: admin-cli
    username: admin
    password: admin

cms:
  ocr:
    chain: groq
    groq-api-key: ${GROQ_API_KEY}
    groq-model: meta-llama/llama-4-scout-17b-16e-instruct
  attachments:
    root-path: ./cms-attachments
    max-file-size: 52428800
    chunk-size: 5242880
    allowed-types: pdf,png,jpg,jpeg,doc,docx,xls,xlsx,txt,csv,zip
    max-files-per-complaint: 10
    temp-dir: temp-chunks
  encryption:
    server-secret: dev-server-secret-key-min16chars
  anti-automation:
    velocity-window-seconds: 60
    velocity-threshold: 100
    min-form-submission-ms: 0
    require-standard-headers: false
  mre:
    version: 1
    re-window-days: 30
    npci-window-days: 30
    card-network-window-days: 60
    filing-deadline-days: 90
    limitation-period-years: 3
    window-basis: BUSINESS
  cors:
    allowed-origins: http://localhost:4200,http://localhost:4201
    allowed-methods: GET,POST,PUT,DELETE,PATCH,OPTIONS
    allowed-headers: Authorization,Content-Type,X-Requested-With,X-Session-Id,X-Form-Start,X-HP-Field,Accept,Origin
    allow-credentials: true
    max-age: 3600
  similar-cases:
    enabled: false
  auth:
    otp:
      length: 6
      expiry-minutes: 10
      max-resend-per-hour: 99
      max-verify-attempts-per-otp: 99
      dev-auto-populate: true
    captcha:
      length: 6
      expiry-minutes: 10
    cooloff:
      progression-seconds: 5,10,20
      max-ceiling-seconds: 20
      reset-after-minutes: 1
    rate-limit:
      otp-requests-per-mobile-per-hour: 99
      otp-requests-per-ip-per-hour: 99
      login-attempts-per-ip-per-minute: 99

management:
  endpoints:
    web:
      exposure:
        include: health,info,metrics
  endpoint:
    health:
      show-details: always

logging:
  level:
    com.hrms.cms: DEBUG
    org.hibernate.SQL: DEBUG

8. Troubleshooting

Error: CMS_ENCRYPTION_SECRET must be set

Fix: Use dev-local or dev-server profile, or set the env var: export CMS_ENCRYPTION_SECRET=my-secret-key-16ch

Error: Keycloak Connection Refused (port 8180)

Fix: Ensure Keycloak is running on port 9090 and application-dev-server.yml has server-url: http://localhost:9090

Error: ORA-12541: TNS:no listener

Fix: Verify Oracle server IP and port. Check network/firewall rules. Ensure Oracle listener is running on server.

Error: Kafka - Connection to node -1 could not be established

Fix: Verify Kafka server IP. Check if Kafka's advertised.listeners on server includes your IP range. Check firewall.

Error: Port 4200 already in use

Fix: Kill existing process: npx kill-port 4200 or use ng serve --port 4202

Error: CORS blocked

Fix: Add your frontend URL to cms.cors.allowed-origins in the application yml file.

Error: "RBIO OFFICER Team" instead of actual user name

Fix: Keycloak is unreachable or no users have RBIO_OFFICER role. Verify Keycloak port in config and that users exist.
